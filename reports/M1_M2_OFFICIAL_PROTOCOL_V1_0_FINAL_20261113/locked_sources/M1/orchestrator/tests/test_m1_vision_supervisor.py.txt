@pytest.mark.parametrize('fault', ['missing', 'schema', 'page', 'revision', 'image_hash', 'c1_hash', 'image_changed'])
def test_round2_C1_rejected_before_backend(rig, fault):
    import hashlib
    c1 = rig[3].run(rig[1], rig[0], 'ocr', lambda n: spec(), rig[-2])
    if fault == 'missing':
        c1 = None
    elif fault == 'revision':
        c1 = replace(c1, identity=replace(c1.identity, input_revision=2))
    elif fault == 'image_hash':
        c1 = replace(c1, image_sha256='0'*64)
    elif fault == 'c1_hash':
        c1 = replace(c1, c1_sha256='0'*64)
    elif fault == 'image_changed':
        rig[-2].write_bytes(b'replaced image')
    else:
        doc = unique_json(c1.documents['c1'])
        doc['schema_version' if fault == 'schema' else 'page_id'] = 'invalid'
        raw = json.dumps(doc).encode()
        c1 = replace(c1, documents={'c1': raw}, c1_sha256=hashlib.sha256(raw).hexdigest())
    rig[2].clear()
    with pytest.raises(VisionError) as caught:
        rig[3].run(rig[1],rig[0],'vlm',lambda n: spec(),rig[-2],c1=c1)
    assert caught.value.code == 1002
    assert not any(e['event']=='started' for e in rig[2])
    assert list(rig[3].root.iterdir()) == []

@pytest.mark.parametrize('fault', ['identity', 'hash', 'object', 'confidence', 'late', 'clothing_null'])

@pytest.mark.parametrize('mode', ['confirmed', 'failed', 'missing', 'late', 'exception'])
def test_round2_backend_cancel_proof(rig, mode):
    identity=rig[0].attempt(rig[1])
    class Backend:
        def settle(self, identity, task_ref, *, cancel):
            if mode=='exception': raise RuntimeError('secret')
            if mode=='missing': return None
            return dict(identity=asdict(replace(identity, attempt_id='old') if mode=='late' else identity),
                task_ref=task_ref, request_result='cancelled' if mode!='failed' else 'failed',
                termination_confirmed=True, confirmed_at=time.time())
    if mode=='confirmed':
        rig[3].settle_backend(rig[1],identity,Backend(),'real-adapter-task-reference',cancel=True)
        assert rig[1].backend_execution['status']=='confirmed'
    else:
        with pytest.raises(VisionError) as caught:
            rig[3].settle_backend(rig[1],identity,Backend(),'real-adapter-task-reference',cancel=True)
        assert caught.value.code==9001 and not retryable(caught.value,'vlm')
        assert rig[1].backend_execution['status']=='backend_execution_unconfirmed'

def spec(mode='ok', *, code=2002, message='transient', transient=False, evidence=None, source='local-fixture'):
    argv = [sys.executable, '-I', str(FAKE.resolve()), '--mode', mode, '--code', str(code), '--message', message]
    if transient:
        argv.append('--transient')
    if evidence:
        argv += ['--evidence', str(evidence)]
    return WorkerSpec(tuple(argv), source)

def test_M1_S17_cancelled_worker_cannot_revive(rig):
    launched = threading.Event()
    old_audit = rig[3].audit
    def audit(event):
        old_audit(event)
        if event['event'] == 'started':
            launched.set()
    rig[3].audit = audit
    errors = []
    def work():
        try:
            rig[3].run(rig[1],rig[0],'ocr',lambda n: spec('sleep'),rig[-2])
        except BaseException as exc:
            errors.append(exc)
    thread = threading.Thread(target=work)
    thread.start()
    assert launched.wait(5)
    rig[0].cancel(rig[1])
    thread.join(10)
    assert not thread.is_alive() and len(errors)==1 and isinstance(errors[0], Cancelled)
    assert rig[1].status == 'cancelled' and rig[1].worker_state == 'reaped'
    assert_drained(rig)

def test_close_idempotent_shared_barrier_rejects_registration(rig):
    launched = threading.Event()
    rig[3].audit = lambda e: (rig[2].append(e), launched.set() if e['event']=='started' else None)
    errors = []
    def work():
        try:
            rig[3].run(rig[1],rig[0],'ocr',lambda n: spec('sleep'),rig[-2])
        except Cancelled as exc:
            errors.append(exc)
    thread=threading.Thread(target=work)
    thread.start()
    assert launched.wait(5)
    shutdowns=[threading.Thread(target=rig[3].close) for _ in range(3)]
    for shutdown in shutdowns:
        shutdown.start()
    for shutdown in shutdowns:
        shutdown.join(10)
        assert not shutdown.is_alive()
    thread.join(1)
    assert not thread.is_alive() and len(errors)==1
    assert_drained(rig)
    rig[3].close()
    with pytest.raises(VisionError,match='shutting down'):
        rig[3].run(rig[1],rig[0],'ocr',lambda n: spec(),rig[-2])

def test_tree_drain_failure_quarantines_and_never_retries(rig,monkeypatch):
    from orchestrator.services.vision_supervisor import supervisor as module
    from orchestrator.services.vision_supervisor.tree import ContainmentError
    import shutil
    original=module.drain
    def failed_drain(*args,**kwargs):
        # Real termination is still performed, then the barrier is fault-injected.
        original(*args,**kwargs)
        raise ContainmentError('injected drain proof failure')
    monkeypatch.setattr(module,'drain',failed_drain)
    with pytest.raises(ContainmentError):
        rig[3].run(rig[1],rig[0],'ocr',lambda n: spec('sleep'),rig[-2])
    assert len(rig[3].quarantined)==1 and rig[1].worker_state=='quarantined'
    identity,tree,work=rig[3].quarantined[0]
    assert work.is_dir() and tree.process.poll() is not None
    assert sum(e['event']=='started' for e in rig[2])==1
    with pytest.raises(ContainmentError):
        rig[3].close()
    # Only this test-owned quarantine, after the actual OS drain above.
    shutil.rmtree(work)

        rig[4].publish(record,record.identity,
            {'json/book01_p03_ocr.json':handoff.c1,'json/book01_p03_vlm.json':b'{"safety":{"pass":false}}',
             'json/book01_p03_refs.json':handoff.sidecar},
            verify=lambda _: pytest.fail('must reject before verifier'))
    assert rig[4].read('book01_p03') is None
