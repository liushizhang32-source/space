def create(c, page='book01_p03', **kw):
    return c.create('local', page, deadline_ms=100000, **kw)


def run(r, j, **kw):
    return run_mock(r[0], j['job_id'], r[2], r[3], r[4], (96, 128), kw.pop('worker', spec()), r[1], **kw)


def acquire(c, j, resource='vision-fixture'):
    i = c.identity(j['job_id']); c.register_execution(i, 'ocr', resource=resource)
    assert c.lease(i)['state'] == 'lease_acquired'
    return i


def no_dispatch(c, i):
    c.observe_execution(i, dict(event='launch_not_dispatched', proof='local-launcher/no-process-created', reason='fixture-before-launch'))
    c.observe_execution(i, dict(event='cleaned'))


def assert_released(c, i):
    assert c.lease(i)['state'] == 'lease_released'
    row = c.db.execute('SELECT * FROM executions WHERE id=?', (i['attempt_id'],)).fetchone()
    assert row['cleaned'] == 1 and row['resource'] is None


def test_T01_acquired_before_worker_failure(rig2):
    c, p, s, g, image = rig2; j = create(c); image.unlink()
    assert c.db.execute('SELECT count(*) FROM executions').fetchone()[0] == 0
    with pytest.raises(FileNotFoundError): run(rig2, j)
    i = c.identity(j['job_id']); assert_released(c, i)
    row = c.db.execute('SELECT * FROM executions').fetchone()
    assert not row['worker_exited'] and not row['backend_terminal_confirmed']
    assert c.lease(i)['reason'] == 'FileNotFoundError' and p.calls == 0
    assert c.get(j['job_id'])['status'] == 'failed'; c.release(j['job_id'])
    # The previously global poisoned slot can now serve another actual chain.
    image.write_bytes(b'fixture'); other = create(c, 'book01_p04')
    assert run(rig2, other)['status'] == 'done'


def test_T02_worker_factory_construction_failure(rig2):
    c, p, s, g, image = rig2; j = create(c)
    r = Generation(j['job_id'], j['page_id'], j['generation_id'], j['revision'], j['epoch'])
    f = RuntimeFences(c, 'ocr'); import threading
    f.pages[r.page_id] = r; f.serial[r.page_id] = threading.Lock()
    events = []
    def audit(e):
        events.append(e); c.observe_execution(asdict(r.identity), e)
    s.audit = audit
    def factory(_): raise ValueError('worker construction failed')
    with pytest.raises(ValueError, match='construction'): s.run(r, f, 'ocr', factory, image)
    assert_released(c, asdict(r.identity)); assert not any(e['event'] == 'started' for e in events)
    assert c.lease(asdict(r.identity))['reason'] == 'ValueError' and p.calls == 0

def test_T08_duplicate_release_idempotent(rig2):

def test_T09_concurrent_jobs_lease_isolation(rig2):
    c, *_ = rig2; jobs = [create(c), create(c, 'book01_p04')]
    def attempt(j):
        try: return acquire(c, j)
        except sqlite3.IntegrityError: return None
    with ThreadPoolExecutor(2) as pool: ids = list(pool.map(attempt, jobs))
    assert sum(i is not None for i in ids) == 1
    winner = next(i for i in ids if i); loser = jobs[ids.index(None)]
    forged = deepcopy(winner); forged['job_id'] = loser['job_id']
    with pytest.raises(ValueError): no_dispatch(c, forged)
    assert c.lease(winner)['state'] == 'lease_acquired'
    no_dispatch(c, winner); next_i = acquire(c, loser); assert c.lease(next_i)['state'] == 'lease_acquired'

def test_receipt_and_resource_release_rollback_together(rig2):
    c, *_ = rig2; j = create(c); i = acquire(c, j)
    c.observe_execution(i, dict(event='launch_not_dispatched', proof='local-launcher/no-process-created', reason='test-rollback'))
    c.db.execute("CREATE TRIGGER deny_release BEFORE UPDATE OF state ON lease001 WHEN NEW.state='lease_released' BEGIN SELECT RAISE(ABORT,'injected commit failure'); END")

    assert c.lease(i)['state'] == 'launch_not_dispatched'
