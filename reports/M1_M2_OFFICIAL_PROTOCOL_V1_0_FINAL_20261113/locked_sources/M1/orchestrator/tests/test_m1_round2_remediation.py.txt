@pytest.mark.parametrize('when',['ocr','vlm','backoff'])
def test_running_cancel_forbids_retry_release_force(rig2,when):
    c,p,s,g,image=rig2; j=create(c); events=[]; cancelled=[]
    def audit(e):
        events.append(e)
        match=(e['event']=='backoff' if when=='backoff' else e['event']=='started' and c.get(j['job_id'])['stages'][when]=='running')
        if match and not cancelled:
            if e['event']=='started':
                witness=ProcessWitness(e['pid']); assert not witness.exited(); witness.close()
            c.cancel(j['job_id']); cancelled.append(True)
            with pytest.raises(ValueError): c.release(j['job_id'])
            with pytest.raises(ValueError): c.create('local','book01_p03',deadline_ms=100000,force=True)
    s.audit=audit
    # Backoff cancellation occurs after the first failed tree has drained. Its
    # local lease may already be terminal; cancellation still forbids retry.
    if when=='backoff':
        def backoff_audit(e):
            events.append(e)
            if e['event']=='backoff': c.cancel(j['job_id']); cancelled.append(True)
        s.audit=backoff_audit
    s.deadline=.25 if when=='backoff' else 5
    with pytest.raises(Cancelled): run(rig2,j,worker=spec('sleep') if when=='backoff' else spec())
    assert cancelled and c.get(j['job_id'])['status']=='cancelled' and p.calls==0
    starts=[e for e in events if e['event']=='started']
    assert len(starts)==(2 if when=='vlm' else 1)
    assert c.read_assets(j['job_id']) is None
    c.release(j['job_id'])
    assert c.create('local','book01_p03',deadline_ms=100000,force=True)['job_id']!=j['job_id']


def test_concurrent_cancel_stops_real_worker(rig2):
    c,p,s,g,image=rig2; j=create(c); started=threading.Event(); cancelled=threading.Event(); errors=[]
    def audit(e):
        if e['event']=='started': started.set(); assert cancelled.wait(3)
    s.audit=audit
    def work():
        try: run(rig2,j,worker=spec('sleep'))
        except Exception as e: errors.append(e)
    worker=threading.Thread(target=work); worker.start(); assert started.wait(5)
    with ThreadPoolExecutor(4) as pool: list(pool.map(lambda _:c.cancel(j['job_id']),range(4)))
    with pytest.raises(ValueError): c.release(j['job_id'])
    cancelled.set(); worker.join(8)
    assert not worker.is_alive() and len(errors)==1 and isinstance(errors[0],Cancelled)
    assert c.db.execute('SELECT count(*) FROM executions').fetchone()[0]==1
    c.release(j['job_id'])


def test_external_client_exit_never_proves_backend_terminal(rig2):
    c,*_=rig2; j=create(c); identity=deepcopy(c.identity(j['job_id']))
    identity['attempt_id']='external-fake'; identity['fencing_token']='external-fake-token'
    c.register_execution(identity,'ocr',scope='external_mock',resource='fake-shared-http')
    c.cancel(j['job_id'])
    c.observe_execution(identity,dict(event='reaped',active=0,member_exit_witness={'fake-client':True}))
    c.observe_execution(identity,dict(event='cleaned'))
    for _ in range(2):
        with pytest.raises(ValueError): c.release(j['job_id'])
        with pytest.raises(ValueError): c.create('local','book01_p03',deadline_ms=100000,force=True)
    with pytest.raises(ValueError): c.confirm_external_mock(identity,dict(mode='mock',identity=identity,terminal=False))
    c.close(); c.__init__(c.path,clock=lambda:1000)
    with pytest.raises(ValueError): c.release(j['job_id'])
    c.confirm_external_mock(identity,dict(mode='mock',identity=identity,terminal=True))
    c.release(j['job_id'])
    assert c.get(j['job_id'])['status']=='cancelled'


@pytest.mark.parametrize('field',['job_id','generation_id','attempt_id','fencing_token','epoch'])
@pytest.mark.parametrize('event',['failure','timeout','cancel','stage'])
def test_all_business_event_identities_fenced(rig2,field,event):
    c,*_=rig2; j=create(c); identity=c.identity(j['job_id']); stale=deepcopy(identity)
    if field=='job_id':
        other=c.create('local','book01_p04',deadline_ms=100000); stale[field]=other['job_id']
        other_before=c.get(other['job_id'])
        assert not c.fail(j['job_id'],'ocr',3004,identity=c.identity(other['job_id']))
        assert c.get(other['job_id'])==other_before
    else: stale[field]=2 if field=='epoch' else 'old'
    target=stale['job_id']; before=c.get(target)
    assert not c.apply(stale,event,stage='ocr',state='running',code=3004)
    assert c.get(target)==before

@pytest.mark.parametrize('window',['submit_before_intent','intent_before_dispatch','provider_after_submit_before_record','recorded_before_return','files_before_db','db_after_publish'])
def test_crash_windows_reuse_same_submission(rig2,window):
    c,p,*_=rig2; j=create(c)
    class Crash(BaseException): pass
    def fault(name):
        if name==window: raise Crash()
    with pytest.raises(Crash): run(rig2,j,fault=fault)
    old=c.db.execute('SELECT idem,task_ref FROM submissions').fetchone()
    before_count=p.db.execute('SELECT count(*) FROM fake_tasks').fetchone()[0]
    c.close(); c.__init__(c.path,clock=lambda:1000)
    with ThreadPoolExecutor(4) as pool: list(pool.map(lambda _:c.recover(),range(4)))
    out=run(rig2,j)
    assert out['status']=='done'
    row=c.db.execute('SELECT * FROM submissions').fetchone()
    assert c.db.execute('SELECT count(*) FROM submissions').fetchone()[0]==1
    assert p.db.execute('SELECT count(*) FROM fake_tasks').fetchone()[0]==1 and p.calls==1
    if old: assert row['idem']==old['idem']
    if before_count: assert row['task_ref']==p.lookup(row['idem'])['task_ref']

@pytest.mark.parametrize('window',['provider_after_submit_before_record','files_before_db','db_after_publish'])
def test_actual_process_death_recovers_persistent_backend_task(rig2,tmp_path,window):
    root=tmp_path/'death'; root.mkdir()
    code='''import os,sys
from pathlib import Path
sys.path.insert(0,str(Path('orchestrator/tests').resolve()))
from test_m1_vision_supervisor import spec
from orchestrator.services.mock_core_round2.core import Core
from orchestrator.services.mock_core_round2.runner import run_mock
from orchestrator.services.mock_core_round2.provider import FakeProvider
from orchestrator.services.vision_supervisor.supervisor import Supervisor
from orchestrator.services.vision_supervisor.gate import Gate
p=Path(sys.argv[1]); c=Core(p/'db.sqlite',clock=lambda:1000); c.configure_budget('local','CNY',100)
j=c.create('local','book01_p03',deadline_ms=100000)
provider=FakeProvider(p/'provider.sqlite'); image=p/'image.png'; image.write_bytes(b'fake')
def fault(w):
 if w==sys.argv[2]: os._exit(19)
run_mock(c,j['job_id'],Supervisor(p/'work',deadline=5),Gate({'local-fixture'}),image,(96,128),spec(),provider,fault=fault)
'''
    result=subprocess.run([sys.executable,'-B','-c',code,str(root),window],capture_output=True,text=True)
    assert result.returncode==19,result.stderr
    c=Core(root/'db.sqlite',clock=lambda:1000); p=FakeProvider(root/'provider.sqlite')
    try:
        jid=c.db.execute('SELECT id FROM jobs').fetchone()[0]
        old=c.db.execute('SELECT idem FROM submissions').fetchone()[0]
        out=run_mock(c,jid,Supervisor(root/'resume-work',deadline=5),Gate({'local-fixture'}),root/'image.png',(96,128),spec(),p)
        assert out['status']=='done' and p.calls==0
        assert c.db.execute('SELECT count(*) FROM submissions').fetchone()[0]==1
        assert c.db.execute('SELECT idem FROM submissions').fetchone()[0]==old
        assert p.db.execute('SELECT count(*) FROM fake_tasks').fetchone()[0]==1
    finally: c.close(); p.close()
