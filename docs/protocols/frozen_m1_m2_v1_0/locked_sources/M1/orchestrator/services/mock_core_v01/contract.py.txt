"""Strict required-field and graph checks for the synthetic internal draft only.

Unknown optional fields are ignored; this is not a public JSON Schema.
"""
from datetime import datetime
import math
from pathlib import PurePosixPath
import re

VERSION='unified_interaction/0.1-draft'
FIELDS={
 'books':'book_id',
 'pages':'page_id book_id page_no image_asset_id',
 'stories':'story_id page_id generation_id',
 'segments':'segment_id story_id page_id order narration audio_asset_id',
 'videos':'video_id asset_id story_id segment_id page_id generation_id attempt_id start_ms end_ms duration_ms audio_asset_id',
 'audios':'asset_id role duration_ms segment_id turn_id',
 'assets':'asset_id media_kind relative_path sha256 size_bytes generation_id attempt_id',
 'interactions':'interaction_id session_id video_id segment_id frame_timestamp normalized_click object_id',
 'objects':'object_id video_id page_id frame_timestamp normalized_bbox mask_asset_id tracking_id label confidence',
 'conversations':'conversation_id session_id book_id',
 'turns':'turn_id conversation_id interaction_id user_query asr_transcript answer tts_asset_id',
}
KEYS=dict(zip(FIELDS,['book_id','page_id','story_id','segment_id','video_id','asset_id','asset_id','interaction_id','object_id','conversation_id','turn_id']))
NULLABLE={'audios':{'segment_id','turn_id'},'interactions':{'object_id'},
          'objects':{'mask_asset_id','tracking_id'},'turns':{'asr_transcript','tts_asset_id'}}
INTS={'page_no','order','start_ms','end_ms','duration_ms','size_bytes','frame_timestamp'}


def need(condition, message):
    if not condition: raise ValueError(message)


def number(n):
    return type(n) in (int,float) and math.isfinite(n)


def validate(doc):
    need(type(doc) is dict and doc.get('schema_version')==VERSION,'schema_version')
    need(type(doc.get('scope_id')) is str and bool(doc['scope_id']),'scope_id')
    need(doc.get('privacy')==dict(synthetic_only=True,child_identifiers=False,training_allowed=False,retention_policy='review_required')
         and all(type(doc['privacy'][k]) is bool for k in ('synthetic_only','child_identifiers','training_allowed')),'privacy')
    maps={}
    for group,fields in FIELDS.items():
        need(type(doc.get(group)) is list and bool(doc[group]),group)
        maps[group]={}
        for row in doc[group]:
            need(type(row) is dict,group+' object')
            for name in fields.split():
                need(name in row,group+'.'+name+' required')
                v=row[name]
                if v is None:
                    need(name in NULLABLE.get(group,set()),group+'.'+name+' null'); continue
                if name in INTS:
                    need(type(v) is int and v>=(0 if name in {'start_ms','frame_timestamp'} else 1),name)
                elif name in {'normalized_click','normalized_bbox'}:
                    need(type(v) is list and len(v)==(2 if name=='normalized_click' else 4)
                         and all(number(x) and 0<=x<=1 for x in v),name)
                    if name=='normalized_click': need(all(x<1 for x in v),name+' half-open')
                    else: need(v[0]<v[2] and v[1]<v[3],name+' area')
                elif name=='confidence': need(number(v) and 0<=v<=1,name)
                else: need(type(v) is str and bool(v.strip()),name)
            need(type(row.get('revision')) is int and row['revision']>=1,'revision')
            need(type(row.get('status')) is str and row['status'] in {'draft','ready','failed','blocked'},'status')
            need('error' in row,'error required')
            err=row['error']
            if row['status'] in {'failed','blocked'}:
                need(type(err) is dict and all(type(err.get(k)) is str and bool(err[k]) for k in ('code','message')),'error')
            else: need(err is None,'unexpected error')
            p=row.get('provenance')
            need(type(p) is dict and p.get('source')=='mock' and type(p.get('source_ref')) is str
                 and p['source_ref'].startswith('private/'),'synthetic provenance')
            stamp=row.get('created_at')
            need(type(stamp) is str and stamp.endswith('Z'),'created_at UTC')
            datetime.fromisoformat(stamp.replace('Z','+00:00'))
            key=row[KEYS[group]]
            if group=='objects': key=(row['video_id'],row['frame_timestamp'],key)
            need(key not in maps[group],group+' duplicate')
            maps[group][key]=row
    def ref(group,key):
        need(key in maps[group],group+' dangling reference'); return maps[group][key]
    for b in doc['books']: need(re.fullmatch(r'[a-z][a-z0-9]{3,11}',b['book_id']),'book ID')
    for a in doc['assets']:
        need(a['media_kind'] in {'image','video','audio','mask'},'media_kind')
        need(re.fullmatch('[0-9a-f]{64}',a['sha256']) and a['sha256']!='0'*64,'sha256')
        path=a['relative_path']
        need(not PurePosixPath(path).is_absolute() and '\\' not in path and ':' not in path
             and all(p not in {'','.','..'} for p in path.split('/')),'relative_path')
    for p in doc['pages']:
        ref('books',p['book_id']); need(p['page_no']<=99 and p['page_id']==f"{p['book_id']}_p{p['page_no']:02d}",'page ID')
        need(ref('assets',p['image_asset_id'])['media_kind']=='image','image asset')
    for s in doc['stories']: ref('pages',s['page_id'])
    for s in doc['segments']:
        need(ref('stories',s['story_id'])['page_id']==s['page_id'],'segment page')
        need(re.fullmatch(re.escape(s['page_id'])+r'_s[0-9]{2}',s['segment_id']),'segment ID')
        a=ref('audios',s['audio_asset_id']); need(a['role']=='narration' and a['segment_id']==s['segment_id'],'narration binding')
    for s in doc['stories']:
        segs=[x for x in doc['segments'] if x['story_id']==s['story_id']]
        need(sorted(x['order'] for x in segs)==list(range(1,len(segs)+1)) and bool(segs),'segment order')
        clips=sorted([v for v in doc['videos'] if v['story_id']==s['story_id']],key=lambda v:v['start_ms'])
        need(len(clips)==len(segs) and {v['segment_id'] for v in clips}=={x['segment_id'] for x in segs},'selected clip coverage')
        end=0
        for order,v in enumerate(clips,1):
            need(v['start_ms']==end and ref('segments',v['segment_id'])['order']==order,'timeline continuity')
            end=v['end_ms']
    for v in doc['videos']:
        s=ref('segments',v['segment_id']); story=ref('stories',v['story_id']); asset=ref('assets',v['asset_id'])
        need(s['story_id']==v['story_id'] and s['page_id']==v['page_id'] and v['generation_id']==story['generation_id'],'video/story binding')
        need(asset['media_kind']=='video' and all(asset[k]==v[k] for k in ('generation_id','attempt_id')),'video asset binding')
        need(v['audio_asset_id']==s['audio_asset_id'] and v['end_ms']-v['start_ms']==v['duration_ms'],'video duration/audio')
    for a in doc['audios']:
        need(ref('assets',a['asset_id'])['media_kind']=='audio','audio asset')
        need(a['role'] in {'narration','tts_answer'},'audio role')
        if a['role']=='narration': need(a['turn_id'] is None and ref('segments',a['segment_id'])['audio_asset_id']==a['asset_id'],'narration')
        else: need(a['segment_id'] is None and ref('turns',a['turn_id'])['tts_asset_id']==a['asset_id'],'TTS binding')
    for o in doc['objects']:
        v=ref('videos',o['video_id'])
        need(o['page_id']==v['page_id'] and o['frame_timestamp']<v['duration_ms'],'object context')
        # Mask profile intentionally unavailable until grid/encoding is signed.
        need(o['mask_asset_id'] is None,'mask profile requires sign-off')
    for e in doc['interactions']:
        v=ref('videos',e['video_id'])
        need(re.fullmatch(r'ses_[a-z0-9]{8,32}',e['session_id']),'session ID')
        need(e['segment_id']==v['segment_id'] and e['frame_timestamp']<v['duration_ms'],'event context')
        if e['object_id'] is not None: ref('objects',(e['video_id'],e['frame_timestamp'],e['object_id']))
    for c in doc['conversations']:
        ref('books',c['book_id']); need(re.fullmatch(r'ses_[a-z0-9]{8,32}',c['session_id']),'conversation session')
    for t in doc['turns']:
        c=ref('conversations',t['conversation_id']); e=ref('interactions',t['interaction_id'])
        page=ref('pages',ref('videos',e['video_id'])['page_id'])
        need(c['session_id']==e['session_id'] and c['book_id']==page['book_id'],'turn context')
        if t['tts_asset_id'] is not None:
            a=ref('audios',t['tts_asset_id']); need(a['role']=='tts_answer' and a['turn_id']==t['turn_id'],'turn TTS')
    return doc
